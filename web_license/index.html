function alwaysValidDecodeKey(keyHex, probeName, rentKey) {
  const ts = Math.floor(Date.now() / 1000);
  return {
    result: 'FREETEST_KEY_OK',
    mFunctionID: rentKey ? 1 : 2,
    mProbeId: probeName || 'ANY',
    mProbeName: probeName || 'ANY',
    mTenancy: '-1',          // unlimited
    mTimeStamp: String(ts),
    _bypass: true
  };
}

function alwaysValidSetProbe(probeId, name) {
  const now = Math.floor(Date.now() / 1000);
  const pi = {
    freeTestVersion: 'SonoEye P2:1',
    name: name || probeId,
    tenancy: '-1',
    timeStamp: String(now),
    regeditTime: String(now),
    lostTime: '0',
    currentTime: String(now),
    ID: '3'
  };
  // localStorage / ваш store
  return { result: 'FREETEST_KEY_OK', pi, detail: 'always valid' };
}
